"""
Say "Hey Jarvis" and an app opens.

Setup:
    pip install SpeechRecognition pyaudio

    (If pyaudio fails to install)
      Windows: pip install pipwin && pipwin install pyaudio
      macOS:   brew install portaudio && pip install pyaudio
      Linux:   sudo apt install portaudio19-dev python3-pyaudio

Run:
    python jarvis.py
"""

import platform
import subprocess
import speech_recognition as sr

# ---- CHANGE THIS to the app you want opened -------------------------------
APP = {
    "Windows": "notepad.exe",      # e.g. "chrome.exe", "calc.exe"
    "Darwin": "Safari",            # e.g. "Spotify", "Calculator" (macOS app name)
    "Linux": "firefox",            # e.g. "gnome-calculator", "code"
}
# ---------------------------------------------------------------------------

WAKE_WORDS = ("hey jarvis", "hey jarvas", "a jarvis", "hey jarvis.")


def open_app():
    system = platform.system()
    app = APP.get(system)
    print(f"Opening {app}...")
    if system == "Windows":
        subprocess.Popen(["cmd", "/c", "start", "", app], shell=False)
    elif system == "Darwin":
        subprocess.Popen(["open", "-a", app])
    else:
        subprocess.Popen([app])


def main():
    recognizer = sr.Recognizer()
    recognizer.dynamic_energy_threshold = True

    with sr.Microphone() as mic:
        print("Calibrating for background noise...")
        recognizer.adjust_for_ambient_noise(mic, duration=1)
        print('Listening... say "Hey Jarvis" (Ctrl+C to quit)')

        while True:
            try:
                audio = recognizer.listen(mic, phrase_time_limit=4)
                text = recognizer.recognize_google(audio).lower()
                print("Heard:", text)

                if any(w in text for w in WAKE_WORDS) or "jarvis" in text:
                    open_app()

            except sr.UnknownValueError:
                pass  # nothing understood, keep listening
            except sr.RequestError as e:
                print("Speech service error (check internet):", e)
            except KeyboardInterrupt:
                print("\nBye!")
                break


if __name__ == "__main__":
    main()"""
